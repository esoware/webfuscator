## What it changes

Converts eligible arrow functions to function expressions while preserving
their lexical captures. Arrows that use `super` or whose values escape stay
unchanged.

## Before and after

<CodeGroup>

```js title="Before" lines
var fn = x => {
  return x + 1;
};
var thisFn = function () {
  var read = () => {
    return this.x;
  };
  return read();
};
var escapes = [() => 1];
class Sub extends Base {
  m() {
    var call = () => super.m();
    return call();
  }
}
```

```js title="After" lines
var fn = function (x) {
  return x + 1;
};
var thisFn = function () {
  var _this = this;
  var read = function () {
    return _this.x;
  };
  return read();
};
var escapes = [() => 1];
class Sub extends Base {
  m() {
    var call = () => super.m();
    return call();
  }
}
```

</CodeGroup>

## Configuration

Run this transform by setting `transforms.arrowToFunction` to `true`:

```js obfuscate.mjs highlight={3}
const output = obfuscate(source, {
  transforms: {
    arrowToFunction: true,
  },
})
```
